All In One - Uploader

Contoh Request via Node.js

Gunakan kode berikut untuk mengunggah file dari server menggunakan axios:

const axios = require('axios');
const axios = require('axios');
const FormData = require('form-data');
const fs = require('fs');

async function uploadFile() {
  const form = new FormData();
  form.append('file', fs.createReadStream('./file-kamu.jpg'));
  form.append('uploader', 'qu.ax');

  try {
    const res = await axios.post(`https://smail.my.id/uploader`, form, {
      headers: form.getHeaders()
    });

    console.log('Sukses:', res.data);
  } catch (err) {
    console.error('Gagal:', err.response?.data || err.message);
  }
}

uploadFile();

Untuk BOT WhatsApp

Gunakan kode berikut untuk mengunggah file lewat bot WhatsApp (type plugins + esm):

import axios from "axios";
import FormData from "form-data";
import fs from "fs";

let handler = async (m, { conn, text, usedPrefix, command }) => {
  let q = m.quoted || m;
  let mime = (q.msg || q).mimetype || '';
  let { data: list } = await axios.get(`https://smail.my.id/uploader/list`);
  let capt = 'Kirim/reply media\n\nPilihan uploader:\n' + list.map((v, i) => `${i + 1}. ${v}`).join('\n') + `\n\nContoh: *${usedPrefix + command} 1*`;

  if (!text || !mime) return m.reply(capt);
  if (text >= 1 && text <= list.length) {
    let type = list[text - 1];
    try {
      m.react("⛅");
      let res = await upload(await q.download(true), type);
      m.reply(Object.entries(res).map(([k, v]) => `*${k}*: ${v}`).join('\n'));
    } catch (e) {
      m.reply("Terjadi kesalahan: " + e.message);
    }
  } else m.reply("Pilihan tidak valid");
};

handler.help = ["aioupload"];
handler.command = /^((aio)?upload(er)?|tourl)$/i;
export default handler;

async function upload(path, type) {
  const form = new FormData();
  form.append('file', fs.createReadStream(path));
  form.append('uploader', type);
  try {
    const { data } = await axios.post(`https://smail.my.id/uploader`, form, {
      headers: form.getHeaders()
    });
    return data;
  } catch (err) {
    return err.response?.data || err.message;
  }
}
